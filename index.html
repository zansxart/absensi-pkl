document.addEventListener('DOMContentLoaded', () => { renderDashboard(); }); async function renderDashboard() { const [students, allAttendance, customSchedules, settings] = await Promise.all([ DB.getStudents(), DB.getAttendance(), DB.getCustomSchedules(), DB.getSettings(), ]); const today = getToday(); const now = new Date(); const year = now.getFullYear(); const month = now.getMonth() + 1; const todayAttendance = allAttendance.filter(a => a.date === today); let hadir = 0, terlambat = 0; const hadirIds = new Set(); todayAttendance.filter(a => a.type === 'masuk').forEach(a => { hadirIds.add(a.studentId); if (a.status === 'warning') terlambat++; else hadir++; }); const alpha = students.filter(s => !hadirIds.has(s.id)).length; const totalHadir = hadirIds.size; document.getElementById('stat-hadir').textContent = hadir; document.getElementById('stat-hadir-sub').textContent = `dari ${students.length} siswa`; document.getElementById('stat-terlambat').textContent = terlambat; document.getElementById('stat-terlambat-sub').textContent = `${terlambat} siswa`; document.getElementById('stat-alpha').textContent = alpha; document.getElementById('stat-alpha-sub').textContent = `${alpha} siswa belum scan`; document.getElementById('stat-total').textContent = students.length; const navWaiting = document.getElementById('nav-waiting'); if (navWaiting) navWaiting.textContent = alpha; const todayList = document.getElementById('today-list'); if (todayAttendance.length === 0) { todayList.innerHTML = `

Belum ada absensi hari ini

`; } else { const grouped = {}; todayAttendance.forEach(a => { if (!grouped[a.studentId]) grouped[a.studentId] = { masuk: null, pulang: null }; grouped[a.studentId][a.type] = a; }); const html = Object.entries(grouped).map(([sid, rec]) => { const s = students.find(st => st.id === sid); if (!s) return ''; const masukTime = rec.masuk ? rec.masuk.time : '-'; const pulangTime = rec.pulang ? rec.pulang.time : '-'; const statusBadge = rec.masuk ? (rec.masuk.status === 'warning' ? `Terlambat` : `Tepat Waktu`) : `Belum Masuk`; const avatarHtml = s.photo ? `` : `
${getInitials(s.name)}
`; return `
${avatarHtml}
${s.name}
${s.school}
${statusBadge}
M: ${masukTime}
P: ${pulangTime}
`; }).join(''); todayList.innerHTML = html || `

Belum ada data

`; } const monthlyBars = document.getElementById('monthly-bars'); if (students.length === 0) { monthlyBars.innerHTML = `

Belum ada data

`; } else { monthlyBars.innerHTML = students.map(s => { const stats = Attendance.getMonthStats(s.id, year, month, allAttendance, students, settings, customSchedules); const pct = stats.pct; const color = pct >= 80 ? 'var(--success)' : pct >= 60 ? 'var(--warning)' : 'var(--danger)'; const avatarHtml = s.photo ? `` : `
${getInitials(s.name)}
`; return `
${avatarHtml} ${s.name} ${pct}%
Hadir: ${stats.hadir} Terlambat: ${stats.terlambat} Alpha: ${stats.alpha}
`; }).join(''); } const tbody = document.getElementById('students-table'); if (students.length === 0) { tbody.innerHTML = `Belum ada siswa terdaftar`; } else { tbody.innerHTML = students.map(s => { const stats = Attendance.getMonthStats(s.id, year, month, allAttendance, students, settings, customSchedules); const avatarHtml = s.photo ? `` : `
${getInitials(s.name)}
`; const statusBadge = hadirIds.has(s.id) ? (todayAttendance.find(a => a.studentId === s.id && a.type === 'masuk')?.status === 'warning' ? `Terlambat` : `Hadir`) : `Belum Hadir`; const todayRec = todayAttendance.find(a => a.studentId === s.id && a.type === 'masuk'); const activeShift = todayRec?.shift || s.shift || 'P'; return `
${avatarHtml}
${s.name}
${s.nis} • ${s.major}
${s.school} ${activeShift === 'P' ? 'Pagi' : 'Siang'} ${statusBadge}
${stats.pct}%
`; }).join(''); } }
